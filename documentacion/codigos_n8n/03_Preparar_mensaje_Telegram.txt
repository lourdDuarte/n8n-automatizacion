const aviso = $json.__aviso || {};
const notif = aviso.notificacion || {};

const limpiar = (v) => String(v ?? '')
  .replace(/[\r\n\t]+/g, ' ')
  .replace(/\s+/g, ' ')
  .trim();

const valor = (v) => {
  if (typeof v === 'number') {
    return new Intl.NumberFormat('es-AR', { maximumFractionDigits: 2 }).format(v);
  }
  if (typeof v === 'string' && v.startsWith('=')) return 'fórmula actualizada';
  return limpiar(v);
};

const extraerMotivo = (r) => {
  const texto = limpiar(r.error_final || r.detalle || r.salida_final || '');
  if (!texto) return 'La fuente informó un error sin detalle adicional.';
  const partes = texto.split(/(?=ERROR(?: FINAL)?:|RESULTADO: ERROR|No se |El período|El IPI)/i);
  return limpiar(partes[partes.length - 1]).slice(0, 900);
};

const hora = aviso.fin
  ? new Date(aviso.fin).toLocaleString('es-AR', {
      timeZone: 'America/Argentina/Buenos_Aires',
      day: '2-digit', month: '2-digit', year: 'numeric',
      hour: '2-digit', minute: '2-digit'
    })
  : new Date().toLocaleString('es-AR', {
      timeZone: 'America/Argentina/Buenos_Aires'
    });

let lineas = [];
if (aviso.estado === 'ERROR' || (aviso.errores || []).length) {
  lineas.push('⚠️ ACTUALIZACIÓN CON ERRORES', '', `Fecha: ${hora}`);
  const fallidos = (aviso.resultados || []).filter(r => r.estado === 'ERROR');
  for (const r of fallidos) {
    lineas.push('', `Indicador: ${limpiar(r.nombre || r.fuente)}`);
    if (r.periodo) lineas.push(`Período: ${limpiar(r.periodo)}`);
    lineas.push(`Motivo: ${extraerMotivo(r)}`);
    lineas.push('Estado: no actualizado.');
  }
  if (!fallidos.length) {
    lineas.push('', `Indicadores con error: ${(aviso.errores || []).join(', ') || 'no identificados'}`);
  }
  lineas.push('', 'Los demás indicadores conservan sus resultados independientes.');
} else {
  const cargas = Array.isArray(aviso.cargas) ? aviso.cargas : [];
  lineas.push('✅ BASE DE DATOS ACTUALIZADA', '', `Fecha: ${hora}`);
  lineas.push(`Indicadores actualizados: ${cargas.length}`);
  lineas.push(`Columnas actualizadas: ${cargas.reduce((n, c) => n + Number(c.columnas_actualizadas || 0), 0)}`);

  for (const c of cargas) {
    const actualizadas = Number(c.columnas_actualizadas || 0);
    const esperadas = Number(c.columnas_esperadas || actualizadas);
    lineas.push('', `📌 ${limpiar(c.nombre || c.fuente)}`);
    if (c.periodo) lineas.push(`Período: ${limpiar(c.periodo)}`);
    lineas.push(`Columnas actualizadas: ${actualizadas} de ${esperadas}`);
    if (actualizadas === esperadas) {
      lineas.push('Estado: actualización completada.');
    } else {
      lineas.push(`Estado: actualización parcial; faltaron ${esperadas - actualizadas} columnas.`);
      if (c.motivo) lineas.push(`Motivo probable: ${limpiar(c.motivo)}`);
      if ((c.columnas_faltantes || []).length && esperadas <= 12) {
        lineas.push(`Faltantes: ${(c.columnas_faltantes || []).map(limpiar).join(', ')}`);
      }
      lineas.push(`Consulta: /porque ${limpiar(c.fuente)}`);
    }

    // En indicadores extensos se informa solamente el total.
    if (esperadas <= 12) {
      for (const cambio of (c.cambios || []).slice(0, 12)) {
        lineas.push(`• ${limpiar(cambio.titulo || cambio.celda)}: ${valor(cambio.antes ?? cambio.anterior)} → ${valor(cambio.despues ?? cambio.nuevo)}`);
      }
    }
  }

  // Compatibilidad con avisos antiguos que no contienen cargas.
  if (!cargas.length && (notif.detalle || []).length) {
    for (const d of notif.detalle) {
      lineas.push('', `📌 ${limpiar(d.nombre || d.fuente)}`);
      if (d.periodo) lineas.push(`Período: ${limpiar(d.periodo)}`);
      lineas.push(`Columnas actualizadas: ${Number(d.celdas_actualizadas || 0)}`);
    }
  }
}

let mensaje = lineas.join('\n');
if (mensaje.length > 3900) {
  mensaje = mensaje.slice(0, 3860) + '\n…mensaje abreviado.';
}
return [{ json: { mensaje } }];
