const original = $('Seleccionar diagnóstico').first().json;
const e = original.execution || {};
const w = original.workflow || {};
const err = e.error || {};

const limpiar = v => String(v ?? '')
  .replace(/[\r\n\t]+/g, ' ')
  .replace(/\s+/g, ' ')
  .trim();

let resumen = null;
try {
  const candidato = JSON.parse(String($json.stdout || '{}').replace(/^\uFEFF/, ''));
  const fin = candidato.fin ? new Date(candidato.fin).getTime() : 0;
  const reciente = fin && Math.abs(Date.now() - fin) <= 30 * 60 * 1000;
  if (reciente) resumen = candidato;
} catch {}

const clasificar = texto => {
  const t = limpiar(texto).toLowerCase();
  if (/encabezado|header|columna.*no.*encontr/.test(t)) return 'Encabezado o columna no encontrada';
  if (/no.*disponible|todavía|aún no|periodo esperado/.test(t)) return 'Publicación todavía no disponible';
  if (/valor.*no.*encontr|no se encontró|faltan/.test(t)) return 'Valor no encontrado en la fuente';
  if (/download|descarg|http|connection|timeout/.test(t)) return 'Error de descarga o conexión';
  if (/google|sheet|row_number|fila/.test(t)) return 'Error al localizar o escribir la fila de destino';
  return 'Error de ejecución';
};

const extraerMotivo = r => {
  const bruto = limpiar(r.error_final || r.detalle || r.salida_final || '');
  if (!bruto) return '';
  const lineas = bruto.split(/(?=ERROR(?: FINAL)?:|RESULTADO: ERROR|No se |El período|El IPI)/i);
  return limpiar(lineas[lineas.length - 1]).slice(0, 1200);
};

const fecha = new Date().toLocaleString('es-AR', {
  timeZone: 'America/Argentina/Buenos_Aires'
});
const lineas = [
  '⚠️ ERROR EN LA AUTOMATIZACIÓN', '',
  `Fecha: ${fecha}`,
  `Workflow: ${limpiar(w.name || 'No identificado')}`,
  `Nodo: ${limpiar(e.lastNodeExecuted || 'No identificado')}`
];

const fallidos = (resumen?.resultados || []).filter(r => r.estado === 'ERROR');
if (fallidos.length) {
  for (const r of fallidos) {
    lineas.push('', `Indicador: ${limpiar(r.nombre || r.fuente)}`);
    if (r.periodo) lineas.push(`Período: ${limpiar(r.periodo)}`);
    const motivo = extraerMotivo(r) || 'La fuente no entregó un detalle adicional.';
    lineas.push(`Tipo: ${clasificar(motivo)}`);
    lineas.push(`Motivo: ${motivo}`);
    lineas.push('Estado: no actualizado.');
  }
} else {
  let detalle = limpiar(err.message || err.description || 'Error no especificado');
  if (detalle.length > 1200) detalle = detalle.slice(0, 1200) + '…';
  lineas.push('', 'Motivo:', detalle);
}

lineas.push('', 'La base conserva los datos anteriores del indicador afectado.');
let mensaje = lineas.join('\n');
if (mensaje.length > 3900) mensaje = mensaje.slice(0, 3860) + '\n…mensaje abreviado.';
return [{ json: { mensaje } }];
